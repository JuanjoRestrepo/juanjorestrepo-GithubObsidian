```Python
class Persona:

    def __init__(self, nombre, edad, nacionalidad):
        self.nombre = nombre
        self.edad = edad
        self.nacionalidad = nacionalidad

    def hablar(self):
        print("Hola hola")

  
  
class Artista:
    def __init__(self, habilidad):
        self.habilidad = habilidad

    def mostrarHabilidad(self):
        print(f"Mi habilidad es: {self.habilidad}")

  
  
class EmpleadoArtista(Persona, Artista):
    def __init__(self, nombre, edad, nacionalidad, habilidad, salario, empresa):
        # Constructor
        Persona.__init__(self, nombre, edad, nacionalidad) # Heredamos de Clase Persona
        Artista.__init__(self, habilidad) # Heredamos de Clase Artista
        self.salario = salario
        self.empresa = empresa

	def presentarse(self):
        return f'{super().mostrarHabilidad()}' # usamos super porque queremos llamar un metodo de la clase padre
```
