#  parseador robusto
def sanitize_number(num_str):
    """
    Convierte cadenas monetarias a float de forma robusta.
    Maneja formatos como:
      "$ 1,234.56" -> 1234.56
      "1.234,56"   -> 1234.56
      "$ 189.75"     -> 189.75
      "189,75"     -> 189.75
    Devuelve None si no pudo parsear o si el texto indica '%' (oferta).
    """
    if num_str is None:
        return None
    s = str(num_str).strip()
    if s == "":
        return None

    # Si hay '%' lo ignoramos aquí (se maneja ofertas en otra parte)
    if "%" in s:
        return None

    # quitar símbolos de moneda y letras (dejamos solo dígitos, puntos, comas, signo)
    s_clean = re.sub(r"[^\d\.,\-]", "", s)

    # casos simples: solo dígitos -> int
    if re.fullmatch(r"-?\d+", s_clean):
        try:
            return float(s_clean)
        except:
            return None

    # Si contiene tanto '.' como ',' decidir cuál es decimal según posición
    if '.' in s_clean and ',' in s_clean:
        # caso típico "1,234.56" -> '.' decimal, "," miles
        if s_clean.rfind('.') > s_clean.rfind(','):
            candidate = s_clean.replace(',', '')
            try:
                return float(candidate)
            except:
                pass
        else:
            # caso "1.234,56" -> ',' decimal
            candidate = s_clean.replace('.', '').replace(',', '.')
            try:
                return float(candidate)
            except:
                pass

    # Sólo comas (ej "1234,56" o "1,234")
    if ',' in s_clean and '.' not in s_clean:
        parts = s_clean.split(',')
        # heurística: si hay exactamente 1 coma y 2 decimales asumimos decimal
        if len(parts) == 2 and len(parts[1]) in (2,):
            candidate = s_clean.replace(',', '.')
            try:
                return float(candidate)
            except:
                pass
        # en otro caso, asumimos comas como miles => eliminar
        candidate = s_clean.replace(',', '')
        try:
            return float(candidate)
        except:
            pass

    # Sólo puntos (ej "1234.56" o "1.234")
    if '.' in s_clean and ',' not in s_clean:
        parts = s_clean.split('.')
        # si tiene exactamente 1 punto y 2 digitos después -> decimal
        if len(parts) == 2 and len(parts[1]) in (2,):
            try:
                return float(s_clean)
            except:
                pass
        # si no, asumir puntos como separador de miles -> eliminar
        candidate = s_clean.replace('.', '')
        try:
            return float(candidate)
        except:
            pass

    # Último recurso: extraer el primer patrón numérico razonable
    m = re.search(r"-?\d+[\,\.]?\d*", s_clean)
    if m:
        candidate = m.group(0).replace(',', '.')
        try:
            return float(candidate)
        except:
            pass

    return None

def sanitize_number(num_str: Optional[str]) -> Optional[float]:
    if not num_str:
        return None
    s = str(num_str).strip()
    s = re.sub(r"[^\d\.,\-]", "", s)
    # handle comma/point conventions
    if s.count(",") and s.count("."):
        if s.rfind(",") > s.rfind("."):
            s = s.replace(".", "").replace(",", ".")
        else:
            s = s.replace(",", "")
    else:
        if s.count(",") == 1 and len(s.split(",")[-1]) == 2:
            s = s.replace(",", ".")
        else:
            s = s.replace(",", "")
    try:
        return float(s)
    except Exception:
        return None

def retry(func: Callable, attempts: int = 3, base_delay: float = 0.6):
    last_exc = None
    for i in range(attempts):
        try:
            return func()
        except Exception as e:
            last_exc = e
            wait = base_delay * (2 ** i) + random.uniform(0, 0.2)
            time.sleep(wait)
    raise last_exc

def random_delay(min_s: float = 1.2, max_s: float = 3.5):
    time.sleep(random.uniform(min_s, max_s))

# 5) Robust currency parser (keep your sanitize_number but ensure it handles None)
def sanitize_number_fixed(x):
    """
    Versión mejorada del parser:
    - Trata como decimal si la parte después del punto/coma tiene 1 o 2 dígitos.
    - Maneja combinaciones de '.' y ',' como miles/decimales correctamente.
    - Saneamiento seguro para None / empty.
    """
    if x is None:
        return None
    s = str(x).strip()
    if s == "" or s.lower() in ['nan','none','null']:
        return None
    # Si contiene % -> no es un price parseable aquí
    if '%' in s:
        return None

    # Keep only digits, dot, comma, minus
    s_clean = re.sub(r"[^\d\.,\-]", "", s)
    if s_clean == "":
        return None

    # Case: both comma and dot present -> decide by last separator position
    try:
        if ',' in s_clean and '.' in s_clean:
            # If last comma is after last dot -> comma is decimal
            if s_clean.rfind(',') > s_clean.rfind('.'):
                normalized = s_clean.replace('.', '').replace(',', '.')
                return float(normalized)
            else:
                # dot is decimal
                normalized = s_clean.replace(',', '')
                return float(normalized)

        # Case: only comma present
        if ',' in s_clean and '.' not in s_clean:
            parts = s_clean.split(',')
            # If last part length 1 or 2 -> treat as decimal
            if len(parts[-1]) in (1,2):
                return float(s_clean.replace(',', '.'))
            else:
                # comma as thousands separator
                return float(s_clean.replace(',', ''))

        # Case: only dot present
        if '.' in s_clean and ',' not in s_clean:
            parts = s_clean.split('.')
            # If last part length 1 or 2 -> decimal (handle '137.0', '4.6', '189.75')
            if len(parts[-1]) in (1,2):
                return float(s_clean)
            else:
                # dot as thousands separator -> remove dots
                return float(s_clean.replace('.', ''))

        # No separators -> integer
        return float(s_clean)
    except Exception as e:
        # fallback: extract first number-like
        m = re.search(r"-?\d+[\,\.]?\d*", s_clean)
        if m:
            try:
                return float(m.group(0).replace(',', '.'))
            except:
                return None
        return None

import re

def sanitize_number_fixed(x):
    """
    Versión mejorada del parser:
    - Trata como decimal si la parte después del punto/coma tiene 1 o 2 dígitos.
    - Maneja combinaciones de '.' y ',' como miles/decimales correctamente.
    - Saneamiento seguro para None / empty.
    """
    if x is None:
        return None
    s = str(x).strip()
    if s == "" or s.lower() in ['nan','none','null']:
        return None
    # Si contiene % -> no es un price parseable aquí
    if '%' in s:
        return None

    # Keep only digits, dot, comma, minus
    s_clean = re.sub(r"[^\d\.,\-]", "", s)
    if s_clean == "":
        return None

    # Case: both comma and dot present -> decide by last separator position
    try:
        if ',' in s_clean and '.' in s_clean:
            # If last comma is after last dot -> comma is decimal
            if s_clean.rfind(',') > s_clean.rfind('.'):
                normalized = s_clean.replace('.', '').replace(',', '.')
                return float(normalized)
            else:
                # dot is decimal
                normalized = s_clean.replace(',', '')
                return float(normalized)

        # Case: only comma present
        if ',' in s_clean and '.' not in s_clean:
            parts = s_clean.split(',')
            # If last part length 1 or 2 -> treat as decimal
            if len(parts[-1]) in (1,2):
                return float(s_clean.replace(',', '.'))

                return float(s_clean.replace(',', ''))

            # If last part length 1 or 2 -> decimal (handle '137.0', '4.6', '189.75')
            if len(parts[-1]) in (1,2):

                return float(s_clean.replace('.', ''))

from playwright.sync_api import TimeoutError as PlayTimeout

def extract_fees_from_cart(page, timeout=10000):
    """
    Simulate add-to-cart and navigate to cart or listen to XHR that returns order summary.
    Returns (delivery_fee, service_fee) floats or (None, None).
    """
    delivery_fee = None
    service_fee = None
    try:
        # Click cart/checkout, wait for network response that contains 'order' or 'cart' pricing
        with page.expect_response(lambda r: 'order' in r.url or 'cart' in r.url, timeout=timeout) as resp_info:
            # try navigate to cart button or click checkout
            # adapt selector to Rappi UI
            try:
                page.click("button[data-testid='go-to-cart']", timeout=3000)
            except:
                # fallback: open cart URL
                page.goto(page.url + "/cart", timeout=5000)
        resp = resp_info.value
        try:
            data = resp.json()
        except:
            body = resp.text()
            data = None
            # try parse JSON inside body if necessary
        if data:
            # heuristics to find fees inside JSON
            # 1) search keys with 'delivery' and 'service'
            import json, re
            json_str = json.dumps(data)
            # extract numeric tokens near 'delivery'
            m_delivery = re.search(r'"delivery.*?([0-9]+[.,]?[0-9]*)', json_str, re.IGNORECASE)
            if m_delivery:
                delivery_fee = float(m_delivery.group(1).replace(',', '.'))
            m_service = re.search(r'"service.*?([0-9]+[.,]?[0-9]*)', json_str, re.IGNORECASE)
            if m_service:
                service_fee = float(m_service.group(1).replace(',', '.'))
    except PlayTimeout:
        # no response found, optionally try direct DOM extraction from cart page
        try:
            # Example selectors — adapt to UI
            text_delivery = page.query_selector("div.fee-delivery").inner_text()
            text_service = page.query_selector("div.fee-service").inner_text()
            # parse numbers with your sanitize_number_fixed
            delivery_fee = sanitize_number_fixed(text_delivery)
            service_fee = sanitize_number_fixed(text_service)
        except:
            pass
    except Exception as e:
        print("extract_fees_from_cart error:", e)
    return delivery_fee, service_fee

                return float(s_clean.replace(',', ''))

        if '.' in s_clean and ',' not in s_clean:
            parts = s_clean.split('.')
            # If last part length 1 or 2 -> decimal (handle '137.0', '4.6', '189.75')
            if len(parts[-1]) in (1,2):
                return float(s_clean)
            else:
                # dot as thousands separator -> remove dots
                return float(s_clean.replace('.', ''))

        # No separators -> integer
        return float(s_clean)
    except Exception as e:
        # fallback: extract first number-like
        m = re.search(r"-?\d+[\,\.]?\d*", s_clean)
        if m:
            try:
                return float(m.group(0).replace(',', '.'))
            except:
                return None
        return None
